// ソフトウェアをアップデート
$ sudo dnf update -y

// 日時を表示
$ date

// 追加リポジトリをインストール
$ sudo dnf install -y https://dl.fedoraproject.org/pub/epel/epel-release-latest-9.noarch.rpm

// cowsay, neofetch をインストール
$ sudo dnf install -y cowsay neofetch

// cowsay コマンドを実行
$ cowsay 'hello Google Cloud!'

// neofetch コマンドを実行
$ neofetch
